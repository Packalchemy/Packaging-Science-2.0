const steps = [

  { title:'Blank Paperboard', desc:'We start with a large, flat sheet of strong paperboard — the raw ingredient of every box.', emoji:'📄', cls:'', btnIcon:'🍪', btnLabel:'Stamp the Cookie Cutter!' },

];

let stepIdx = 0;

const mfgPaper = document.getElementById('mfgPaper');

const mfgEmoji = document.getElementById('mfgEmoji');

const mfgTitle = document.getElementById('mfgTitle');

const mfgDesc  = document.getElementById('mfgDesc');

const mfgBtnIcon  = document.getElementById('mfgBtnIcon');

const mfgBtnLabel = document.getElementById('mfgBtnLabel');

const dots = [0,1,2].map(i=>document.getElementById('dot'+i));

const lbls = [0,1,2].map(i=>document.getElementById('lbl'+i));

const lines= [0,1].map(i=>document.getElementById('line'+i));


function applyStep(i) {

  const s = steps[i];

  mfgPaper.className = 'mfg-paper' + (s.cls ? ' '+s.cls : '');

  mfgEmoji.textContent     = s.emoji;

  mfgTitle.textContent     = s.title;

  mfgDesc.textContent      = s.desc;

  mfgBtnIcon.textContent   = s.btnIcon;

  mfgBtnLabel.textContent  = s.btnLabel;

  dots.forEach((d,di)=>{

    d.className='step-circle';

    lbls[di].className='step-lbl';

    if(di<i)  d.classList.add('done');

    if(di===i){ d.classList.add('active'); lbls[di].classList.add('active'); }

  });

  lines.forEach((l,li)=> l.classList.toggle('filled', li<i));

}

applyStep(0);

document.getElementById('mfgBtn').addEventListener('click', ()=>{

  stepIdx = (stepIdx+1)%steps.length;

  applyStep(stepIdx);

});
